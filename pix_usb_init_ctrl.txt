void usb_wait(ADDRESS address = USB_FLAG_CSR, VALUE flag = USB_FLAG_BUSY, VALUE invert = 0, VALUE flip = 0)
{
    if (!flip)
    {
        while ((invert) ? (mmio_read(address) & flag) : (!(mmio_read(address) & flag)));
    }
    else
    {
        while ((invert) ? (usb_read(address) & flag) : (!(usb_read(address) & flag)));
    }
}

    mmio_write(USB_FLAG_GEN, 0);
    usb_wait();
}
